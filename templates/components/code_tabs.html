{% component code_tabs(id, label, tabs, combined_label="") -%} {# Each tab owns a fenced block, so labels and languages belong to the author. Trim template whitespace so Markdown cannot treat indented HTML as code. #} {%- set parts = body | trim | split(pat="") -%} {%- set tab_count = tabs | length -%} {%- if tab_count == 0 or parts | length != tab_count -%} {{ throw(message="code_tabs requires one label per block, separated by ") }} {%- endif -%} {%- set view_count = tab_count -%} {%- set fallback = 0 -%} {%- if combined_label -%} {%- set view_count = tab_count + 1 -%} {%- set fallback = tab_count -%} {# Join adjacent fences with a blank line, retaining the first block's language. -#} {%- set combined = body | trim | regex_replace(pattern="\n[[:space:]]*```[[:space:]]*[[:space:]]*```[^\n]*\n", rep="\n\n") -%} {%- endif -%}
{# Without JavaScript, show the combined view or all individually labeled blocks. -#}
{%- for index in range(end=view_count) -%} {%- if index < tab_count -%}{% set name = tabs[index] %}{%- else -%}{% set name = combined_label %}{%- endif %} {%- endfor %}
{%- for index in range(end=view_count) -%} {%- if index < tab_count -%}{% set source = parts[index] %}{%- else -%}{% set source = combined %}{%- endif %}
{%- if not combined_label %}
{{ tabs[index] }}
{%- endif -%} {{- source | trim | markdown | safe -}}
{%- endfor %}
{% endcomponent %}
